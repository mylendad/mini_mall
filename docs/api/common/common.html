common

Общая библиотека common.

Строго инфраструктурные компоненты для микросервисов платформы: конфигурация, логирование, middleware трассировки, Prometheus-метрики, стандартный формат ошибок и конверт событий.

Сервисы подключают common как обычную установленную зависимость и не должны размещать здесь доменную логику (см. спецификацию foundation).

class PrometheusMetricsMiddleware(starlette.middleware.base.BaseHTTPMiddleware): 
25class PrometheusMetricsMiddleware(BaseHTTPMiddleware):
26    """Считает количество запросов и их задержку.
27
28    Метки: ``method``, ``endpoint`` и (для счётчика) ``status_code``.
29    Добавьте в приложение как обычный middleware; метрики отдаются через
30    ``/metrics`` (см. :func:`common.observability` и шаблон сервиса).
31    """
32
33    async def dispatch(self, request: Request, call_next) -> Response:
34        """Замеряет время обработки запроса и обновляет метрики.
35
36        Параметры:
37            request: Входящий HTTP-запрос.
38            call_next: Следующий обработчик в цепочке middleware.
39
40        Возвращает:
41            HTTP-ответ (метрики обновляются до возврата).
42        """
43        method = request.method
44        path = request.url.path
45        start_time = time.time()
46
47        response = await call_next(request)
48
49        duration = time.time() - start_time
50        status_code = str(response.status_code)
51
52        REQUEST_COUNT.labels(method=method, endpoint=path, status_code=status_code).inc()
53        REQUEST_LATENCY.labels(method=method, endpoint=path).observe(duration)
54
55        return response

class RequestIDMiddleware(starlette.middleware.base.BaseHTTPMiddleware): 
15class RequestIDMiddleware(BaseHTTPMiddleware):
16    """Пробрасывает или генерирует ``request_id`` и ``correlation_id``.
17
18    Принимает ``X-Request-ID`` и ``X-Correlation-ID`` из заголовков запроса;
19    при отсутствии генерирует новые значения. Идентификаторы связываются с
20    контекстом structlog и возвращаются в заголовках ответа.
21    """
22
23    async def dispatch(self, request: Request, call_next) -> Response:
24        """Обрабатывает запрос, связывая контекст и замеряя время.
25
26        Параметры:
27            request: Входящий HTTP-запрос.
28            call_next: Следующий обработчик в цепочке middleware.
29
30        Возвращает:
31            HTTP-ответ с заголовками ``X-Request-ID`` и ``X-Correlation-ID``.
32        """
33        request_id = request.headers.get("X-Request-ID") or str(uuid.uuid4())
34        correlation_id = request.headers.get("X-Correlation-ID") or str(uuid.uuid4())
35
36        structlog.contextvars.clear_contextvars()
37        structlog.contextvars.bind_contextvars(
38            request_id=request_id,
39            correlation_id=correlation_id,
40        )
41
42        response = await call_next(request)
43        response.headers["X-Request-ID"] = request_id
44        response.headers["X-Correlation-ID"] = correlation_id
45        return response

def setup_logging(log_level: str = 'INFO') -> None: 
 9def setup_logging(log_level: str = "INFO") -> None:
10    """Настраивает базовое логирование и structlog для JSON-вывода в stdout.
11
12    Формат вывода: JSON-строки с полями ``event``, ``level``, ``timestamp``.
13    Вызов идемпотентный; определён в начале жизненного цикла приложения.
14
15    Параметры:
16        log_level: Уровень логирования (например, ``"INFO"``).
17    """
18    logging.basicConfig(
19        format="%(message)s",
20        stream=sys.stdout,
21        level=getattr(logging, log_level.upper(), logging.INFO),
22    )
23
24    structlog.configure(
25        processors=[
26            structlog.contextvars.merge_contextvars,
27            structlog.processors.add_log_level,
28            structlog.processors.TimeStamper(fmt="iso"),
29            structlog.processors.JSONRenderer(),
30        ],
31        wrapper_class=structlog.make_filtering_bound_logger(
32            getattr(logging, log_level.upper(), logging.INFO)
33        ),
34        cache_logger_on_first_use=True,
35    )

Настраивает базовое логирование и structlog для JSON-вывода в stdout.

Формат вывода: JSON-строки с полями event, level, timestamp. Вызов идемпотентный; определён в начале жизненного цикла приложения.

Параметры: log_level: Уровень логирования (например, "INFO").
